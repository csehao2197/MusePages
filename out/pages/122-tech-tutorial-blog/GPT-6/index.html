● ● ●format-date.js
// A small function. A useful skill.
function formatDate(value) {
  return new Intl.DateTimeFormat(
    'en-US', { dateStyle: 'medium' }
  ).format(new Date(value));
}

formatDate('2026-11-12T12:00:00Z');
→ "Nov 12, 2026" · illustrative output
